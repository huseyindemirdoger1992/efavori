// =====================================================================
// _ApplicationConnectionDb.OnModelCreating içine eklenecek yapılandırma
// (Ürün modülü: indeksler, decimal hassasiyeti, owned type tanımları)
// =====================================================================

// --- Decimal hassasiyetleri (fiyat alanları) ---
modelBuilder.Entity<data._Product.ProductPrices>(e =>
{
    e.Property(p => p.Price).HasPrecision(18, 2);
    e.Property(p => p.DiscountedPrice).HasPrecision(18, 2);
    e.Property(p => p.CostPrice).HasPrecision(18, 2);

    // Güncel fiyat sorgusu (EffectiveTo == null) için kritik indeks
    e.HasIndex(p => new { p.VariantId, p.Currency, p.EffectiveTo });
    e.HasIndex(p => p.ProductId);
});

modelBuilder.Entity<data._Product.ProductVariants>(e =>
{
    e.Property(p => p.WeightKg).HasPrecision(10, 3);
    e.Property(p => p.WidthCm).HasPrecision(10, 2);
    e.Property(p => p.HeightCm).HasPrecision(10, 2);
    e.Property(p => p.LengthCm).HasPrecision(10, 2);

    e.HasIndex(p => p.ProductId);
    e.HasIndex(p => p.Sku);     // SKU araması — entegrasyonlarda en sık kullanılan anahtar
    e.HasIndex(p => p.Barcode); // Barkod araması
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.ProductBundleItems>(e =>
{
    e.Property(p => p.DiscountRate).HasPrecision(5, 2);
    e.HasIndex(p => p.BundleProductId);
});

// --- Products ---
modelBuilder.Entity<data._Product.Products>(e =>
{
    e.HasIndex(p => p.StoreId);
    e.HasIndex(p => p.UserId);
    e.HasIndex(p => new { p.PublishStatus, p.IsActive }); // Vitrin sorguları
    e.OwnsOne(p => p.Meta);
    e.OwnsOne(p => p.Interaction);
    e.OwnsOne(p => p.IsDeleted);
});

// --- Kategori eşleme ---
modelBuilder.Entity<data._Product.ProductCategories>(e =>
{
    e.HasIndex(p => new { p.ProductId, p.CategoryId }).IsUnique(); // Aynı kategori iki kez eklenemez
    e.HasIndex(p => p.CategoryId); // Kategoriye göre ürün listeleme
});

// --- Özellik sistemi ---
modelBuilder.Entity<data._Product.ProductAttributes>(e =>
{
    e.HasIndex(p => new { p.UserId, p.StoreId }); // Sahiplik filtreleri (null = sistem)
    e.HasIndex(p => p.Code);
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.ProductAttributeValues>(e =>
{
    e.HasIndex(p => p.AttributeId);
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.ProductAttributeMappings>(e =>
{
    e.HasIndex(p => new { p.ProductId, p.AttributeId }).IsUnique();
});

modelBuilder.Entity<data._Product.ProductVariantValues>(e =>
{
    e.HasIndex(p => p.VariantId);
    e.HasIndex(p => new { p.VariantId, p.AttributeId }).IsUnique(); // Bir varyantta aynı özellik tek değer
    e.HasIndex(p => p.AttributeValueId); // "Kırmızı olan tüm varyantlar" sorgusu
});

modelBuilder.Entity<data._Product.ProductSpecifications>(e =>
{
    e.HasIndex(p => p.ProductId);
});

// --- Şablon sistemi ---
modelBuilder.Entity<data._Product.AttributeTemplates>(e =>
{
    e.HasIndex(p => new { p.TemplateGroupId, p.Version }).IsUnique(); // Grup içinde sürüm tekildir
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.AttributeTemplateItems>(e =>
{
    e.HasIndex(p => new { p.TemplateId, p.AttributeId }).IsUnique();
});

modelBuilder.Entity<data._Product.AttributeTemplateCategories>(e =>
{
    e.HasIndex(p => new { p.TemplateGroupId, p.CategoryId }).IsUnique();
    e.HasIndex(p => p.CategoryId); // Kategoriden şablon bulma (ürün ekleme sihirbazı 1. adım)
});

// --- Stok ---
modelBuilder.Entity<data._Product.ProductStocks>(e =>
{
    e.HasIndex(p => new { p.VariantId, p.WarehouseId }).IsUnique(); // Varyant+depo tek satır
    e.HasIndex(p => p.ProductId);
    e.HasIndex(p => p.WarehouseId);
});

// --- Varyant görsel grupları ---
modelBuilder.Entity<data._Product.ProductImageVariantGroups>(e =>
{
    e.HasIndex(p => new { p.ProductId, p.AttributeValueId }).IsUnique(); // Ürün+değer tek grup
});

// --- Marketplace katmanı ---
modelBuilder.Entity<data._Product.Marketplaces>(e =>
{
    e.HasIndex(p => p.Code).IsUnique();
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.MarketplaceAttributeMappings>(e =>
{
    e.HasIndex(p => new { p.MarketplaceId, p.AttributeId, p.CategoryId }).IsUnique();
});

modelBuilder.Entity<data._Product.MarketplaceCategoryMappings>(e =>
{
    e.HasIndex(p => new { p.MarketplaceId, p.CategoryId }).IsUnique();
});

modelBuilder.Entity<data._Product.ProductMarketplaceListings>(e =>
{
    e.HasIndex(p => new { p.MarketplaceId, p.ExternalProductId });
    e.HasIndex(p => p.ProductId);
});

// --- Diğer ---
modelBuilder.Entity<data._Product.Brands>(e =>
{
    e.HasIndex(p => p.Slug);
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.ProductSeo>(e =>
{
    e.HasIndex(p => p.Slug).IsUnique(); // SEO URL benzersizliği
    e.HasIndex(p => p.ProductId);
});

modelBuilder.Entity<data._Product.ProductImportProfiles>(e =>
{
    e.HasIndex(p => p.StoreId);
    e.OwnsOne(p => p.IsDeleted);
});

modelBuilder.Entity<data._Product.ProductExternalMedias>().HasIndex(p => p.ProductId);
modelBuilder.Entity<data._Product.ProductDigitalAssets>().HasIndex(p => p.ProductId);
